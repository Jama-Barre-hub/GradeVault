# Production dependencies — what the running application needs.
# Install with:  pip install -r requirements.txt
#
# Only direct dependencies are listed. pip resolves the rest
# (asgiref, sqlparse) automatically.

Django==6.1
python-dotenv==1.2.3

# Django needs a timezone database. Linux servers usually provide one,
# Windows does not, so ship it explicitly for consistent behaviour.
tzdata==2026.3

# ---------- Production serving ----------
# Only used when DEBUG is off. Harmless locally.

# Runs the application. Django's runserver is for development only and
# is explicitly not safe to expose.
gunicorn==23.0.0

# Serves static files from the application process, so no separate web
# server is needed on a small deployment.
whitenoise==6.11.0

# Reads DATABASE_URL, so the database is configuration rather than code.
dj-database-url==3.0.1

# PostgreSQL driver. SQLite is fine while developing; production is not.
psycopg[binary]==3.2.12

# Limits repeated failed sign-ins. Security, so it belongs in production
# rather than in the development extras.
django-axes==8.3.1

# ---------- Report cards ----------

# Draws report cards as PDFs so a school can attach or file one, which
# printing cannot do. Pure Python: no Pango, no Cairo, nothing for the
# deployment to be missing. WeasyPrint would render the existing HTML
# and never drift from it, but it needs system libraries and a failed
# deploy costs more than a hand-drawn layout does.
fpdf2==2.8.8
